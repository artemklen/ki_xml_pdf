cmake_minimum_required(VERSION 3.18)
project(ki_xml_pdf_external_consumer LANGUAGES C CXX)

if(KI_LINK_KIND STREQUAL "static")
    set(ki_xml_pdf_USE_STATIC_LIBS ON)
    set(expected_type STATIC_LIBRARY)
else()
    set(expected_type SHARED_LIBRARY)
endif()
find_package(ki_xml_pdf ${KI_EXPECTED_VERSION} EXACT CONFIG REQUIRED)
foreach(target IN ITEMS ki_xml_pdf::ki_xml_pdf ki_xml_pdf::static ki_xml_pdf::shared)
    if(NOT TARGET ${target})
        message(FATAL_ERROR "Missing installed target ${target}")
    endif()
endforeach()
get_target_property(selected_target ki_xml_pdf::ki_xml_pdf INTERFACE_LINK_LIBRARIES)
get_target_property(actual_type ${selected_target} TYPE)
if(NOT actual_type STREQUAL expected_type)
    message(FATAL_ERROR "Selected ${actual_type}; expected ${expected_type}")
endif()
add_executable(client report.${KI_SOURCE_SUFFIX})
target_compile_definitions(client PRIVATE KI_EXPECTED_VERSION="${KI_EXPECTED_VERSION}")
target_link_libraries(client PRIVATE ki_xml_pdf::ki_xml_pdf)
